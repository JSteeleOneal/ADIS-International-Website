export interface CaseStudyRow {
  metric: string;
  baseline: string;
  drone: string;

  advantage: "drone" | "baseline";
}

  title: string;
  product: string;
  /** What the drone is measured against. */
  baselineLabel: string;

  highlights?: string[];
  note?: string;
}
